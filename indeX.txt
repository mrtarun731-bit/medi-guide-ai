    async function processFile(file) {
        if (!file) return;
        const bar = document.getElementById("bar");
        const bi = document.getElementById("bi");
        const status = document.getElementById("status");
        
        bar.style.display = "block";
        bi.style.width = "10%";
        status.innerHTML = "<i class='fa-solid fa-spinner fa-spin'></i> Reading document...";
        document.getElementById("app").style.display = "none";

        try {
            if (file.type === "application/pdf") {
                const arrayBuffer = await file.arrayBuffer();
                const pdf = await pdfjsLib.getDocument({data: arrayBuffer}).promise;
                let text = "";
                
                for (let i = 1; i <= pdf.numPages; i++) {
                    const page = await pdf.getPage(i);
                    const content = await page.getTextContent();
                    let pageText = content.items.map(item => item.str).join(" ");
                    text += pageText + "\n";
                }

                if (text.trim().length < 20) {
                    status.innerHTML = "<i class='fa-solid fa-eye fa-bounce'></i> Scanned PDF detected. Running OCR (20-30s)...";
                    text = "";
                    for (let i = 1; i <= pdf.numPages; i++) {
                        const page = await pdf.getPage(i);
                        const viewport = page.getViewport({scale: 2.0});
                        const canvas = document.createElement('canvas');
                        const context = canvas.getContext('2d');
                        canvas.height = viewport.height;
                        canvas.width = viewport.width;
                        await page.render({canvasContext: context, viewport: viewport}).promise;
                        
                        const ret = await Tesseract.recognize(canvas, 'eng', {
                            logger: m => {
                                if(m.status === 'recognizing text') bi.style.width = (m.progress * 80 + 10) + "%";
                            }
                        });
                        text += ret.data.text + "\n";
                    }
                }
                extractedText = text;
            } else if (file.type.startsWith("image/")) {
                status.innerHTML = "<i class='fa-solid fa-eye fa-bounce'></i> Running OCR on image (10-20s)...";
                const { data: { text } } = await Tesseract.recognize(
                    file,
                    'eng',
                    {
                        logger: m => {
                            if (m.status === 'recognizing text') bi.style.width = (m.progress * 80 + 10) + "%";
                        }
                    }
                );
                extractedText = text;
            } else if (file.type === "text/plain" || file.name.endsWith(".txt")) {
                status.innerHTML = "<i class='fa-solid fa-file-lines fa-bounce'></i> Reading Text File...";
                extractedText = await file.text();
                bi.style.width = "100%";
            } else {
                throw new Error("Unsupported file type. Please upload PDF, Image, or TXT.");
            }

            if (extractedText.trim().length < 20) throw new Error("Could not read text. File might be empty or blurry.");

            bi.style.width = "100%";
            status.innerHTML = "<i class='fa-solid fa-circle-check' style='color:var(--good)'></i> Analysis Complete!";
            setTimeout(() => { bar.style.display = "none"; }, 800);
            
            document.getElementById("app").style.display = "block";
            document.getElementById("app").scrollIntoView({ behavior: 'smooth' });
            
            extractMedicalData();
            renderRisk();
            renderSummary();
            renderRecs();

        } catch (error) {
            console.error(error);
            status.innerHTML = "❌ Error: " + error.message;
            status.style.color = "var(--danger)";
            bar.style.display = "none";
        }
    }

    // --- FIXED, ULTRA-ACCURATE EXTRACTION ENGINE ---
    function extractMedicalData() {
        structuredFindings = [];
        let cleanText = extractedText.replace(/[ \t]+/g, ' ');

        // The new Regex: 
        // 1. Finds the test name.
        // 2. Skips up to 25 characters that are NOT digits or hyphens (this skips words like "is", ":", or spaces).
        // 3. Captures the floating-point number.
        const testPatterns = [
            { name: "TSH", regex: /(TSH|Thyroid Stimulating Hormone)[^\d-]{0,25}?(\d+(?:\.\d+)?)/i, min: 0.4, max: 4.5 },
            { name: "T3", regex: /(T3|Triiodothyronine)[^\d-]{0,25}?(\d+(?:\.\d+)?)/i, min: 80, max: 200 },
            { name: "T4", regex: /(T4|Thyroxine)[^\d-]{0,25}?(\d+(?:\.\d+)?)/i, min: 5.0, max: 12.0 },
            { name: "Total Cholesterol", regex: /(Total Cholesterol|Cholesterol Total|Cholesterol)[^\d-]{0,25}?(\d+(?:\.\d+)?)/i, min: 100, max: 200 },
            { name: "LDL Cholesterol", regex: /(LDL|Low Density)[^\d-]{0,25}?(\d+(?:\.\d+)?)/i, min: 0, max: 100 },
            { name: "HDL Cholesterol", regex: /(HDL|High Density)[^\d-]{0,25}?(\d+(?:\.\d+)?)/i, min: 40, max: 100 },
            { name: "Triglycerides", regex: /(Triglycerides|TG)[^\d-]{0,25}?(\d+(?:\.\d+)?)/i, min: 0, max: 150 },
            { name: "Blood Sugar", regex: /(Glucose|Blood Sugar|FBS|Fasting)[^\d-]{0,25}?(\d+(?:\.\d+)?)/i, min: 70, max: 100 },
            { name: "HbA1c", regex: /(HbA1c|Glycated)[^\d-]{0,25}?(\d+(?:\.\d+)?)/i, min: 4.0, max: 5.7 },
            { name: "Hemoglobin", regex: /(Hemoglobin|Hb)[^\d-]{0,25}?(\d+(?:\.\d+)?)/i, min: 13.5, max: 17.5 },
            { name: "Vitamin D", regex: /(Vitamin D|25-OH)[^\d-]{0,25}?(\d+(?:\.\d+)?)/i, min: 30, max: 100 },
            { name: "Creatinine", regex: /(Creatinine)[^\d-]{0,25}?(\d+(?:\.\d+)?)/i, min: 0.6, max: 1.3 }
        ];

        testPatterns.forEach(test => {
            const match = cleanText.match(test.regex);
            if (match) {
                const val = parseFloat(match[2]);
                
                // Check both text clues and mathematical thresholds!
                const startIdx = Math.max(0, match.index - 10);
                const endIdx = Math.min(cleanText.length, match.index + match[0].length + 15);
                const context = cleanText.substring(startIdx, endIdx).toLowerCase();
                
                const hasTextClue = /high|low|abnormal|deficient|elevated|borderline|positive|prediabetes/i.test(context);
                const isMathAbnormal = (val < test.min || val > test.max);

                const isAbnormal = hasTextClue || isMathAbnormal;
                
                structuredFindings.push({ name: test.name, value: val, isAbnormal: isAbnormal });
            }
        });
    }

    function renderRisk() {
        const out = document.getElementById("risk-text");
        const abnormals = structuredFindings.filter(f => f.isAbnormal);
        if (abnormals.length === 0) {
            out.innerHTML = `<div class="data-card normal"><div class="data-title">All Clear</div><div class="data-value" style="font-size:20px;">No critical abnormalities detected</div></div>`;
            return;
        }
        out.innerHTML = abnormals.map(f => `
            <div class="data-card abnormal">
                <div class="data-title">${f.name}</div>
                <div class="data-value">${f.value} <span class="data-badge">High Risk</span></div>
            </div>`).join('');
    }

    function renderSummary() {
        const out = document.getElementById("summary-text");
        if (structuredFindings.length === 0) {
            out.innerHTML = `<p>No standard lab tests detected in this document.</p>`;
            return;
        }
        out.innerHTML = structuredFindings.map(f => `
            <div class="data-card ${f.isAbnormal ? 'abnormal' : 'normal'}">
                <div class="data-title">${f.name}</div>
                <div class="data-value">${f.value} <span class="data-badge">${f.isAbnormal ? 'Abnormal' : 'Normal'}</span></div>
            </div>`).join('');
    }

            recs.push({ icon: "fa-candy-cane", text: "<b>Blood Sugar:</b> Monitor carbohydrate intake. Avoid sugary drinks and refined carbs." });
        }
        if (text.includes("vitamin d")) {

        if (text.includes("tsh") || text.includes("thyroid")) {
            recs.push({ icon: "fa-shield-virus", text: "<b>Thyroid:</b> Ensure adequate iodine and selenium intake. Consult an endocrinologist." });
        }

        if (recs.length === 0) recs.push({ icon: "fa-heart-pulse", text: "Maintain a balanced diet, stay hydrated, and continue regular check-ups." });
        
        out.innerHTML = recs.map(r => `<div class="rec-item"><i class="fa-solid ${r.icon}"></i><div class="rec-text">${r.text}</div></div>`).join('') + 
            `<div style="padding:20px; text-align:center; color:var(--danger); font-size:13px; margin-top:20px;"><i class="fa-solid fa-circle-exclamation"></i> Disclaimer: This AI analysis is for educational purposes. Always consult a licensed physician.</div>`;
    }
